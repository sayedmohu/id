function setupPreview(inputId, imgId, placeholderId) {
const input = document.getElementById(inputId);
const img = document.getElementById(imgId);
const placeholder = document.getElementById(placeholderId);
input.addEventListener('change', function () {
const file = input.files[0];
if (!file) {
img.removeAttribute('src');
img.classList.remove('loaded');
placeholder.style.display = '';
return;
}
if (!file.type.startsWith('image/')) {
alert('Please select an image file.');
input.value = '';
img.removeAttribute('src');
img.classList.remove('loaded');
placeholder.style.display = '';
return;
}
const reader = new FileReader();
reader.onload = function (e) {
img.src = e.target.result;
img.classList.add('loaded');
placeholder.style.display = 'none';
};
reader.readAsDataURL(file);
});
}
setupPreview('topImage', 'topPreview', 'topPlaceholder');
setupPreview('bottomImage', 'bottomPreview', 'bottomPlaceholder');
async function generatePDF() {
const nameInput = document.getElementById('pdfName').value.trim();
const topFile = document.getElementById('topImage').files[0];
const bottomFile = document.getElementById('bottomImage').files[0];
if (!nameInput) { alert('Please enter a name for the PDF.'); return; }
if (!topFile && bottomFile) { alert('Please upload the front image before generating the PDF.'); return; }
if (!topFile && !bottomFile) { alert('Please upload at least one image.'); return; }
const unit = 'mm';
const a4Width = 210;
// Vertical ID card dimensions
const cardWidth = 63;
const cardHeight = 100;
const spacing = 20; // horizontal gap between cards
const topMargin = 50; // small space from top of page
const { jsPDF } = window.jspdf;
const doc = new jsPDF({ unit, format: 'a4' });
const toDataURL = file =>
new Promise(resolve => {
const reader = new FileReader();
reader.onload = () => resolve(reader.result);
reader.readAsDataURL(file);
});
let topDataURL = null;
let bottomDataURL = null;
if (topFile) topDataURL = await toDataURL(topFile);
if (bottomFile) bottomDataURL = await toDataURL(bottomFile);
if (topDataURL && bottomDataURL) {
const totalWidth = (cardWidth * 2) + spacing;
const startX = (a4Width - totalWidth) / 2;
const rightX = startX + cardWidth + spacing;
doc.addImage(topDataURL, 'PNG', startX, topMargin, cardWidth, cardHeight, undefined, 'FAST');
doc.addImage(bottomDataURL, 'PNG', rightX, topMargin, cardWidth, cardHeight, undefined, 'FAST');
} else if (topDataURL) {
const startX = (a4Width - cardWidth) / 2;
doc.addImage(topDataURL, 'PNG', startX, topMargin, cardWidth, cardHeight, undefined, 'FAST');
}
const now = new Date();
const timestamp = String(now.getDate()).padStart(2, '0') +
String(now.getMonth() + 1).padStart(2, '0') +
now.getFullYear().toString() + '_' +
String(now.getHours()).padStart(2, '0') +
String(now.getMinutes()).padStart(2, '0') +
String(now.getSeconds()).padStart(2, '0');
const safeName = nameInput.replace(/[^a-zA-Z0-9_\- ]/g, '').replace(/\s+/g, '_');
const filename = `${safeName}_${timestamp}.pdf`;
doc.save(filename);
alert('PDF generated and downloaded successfully!');
}
